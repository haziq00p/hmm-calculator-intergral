    // Mathematical parser & evaluator
    function compileFunction() {
      const expr = functionInput.value.trim();
      if (!expr) {
        showError("Please enter a valid mathematical expression.");
        return false;
      }
      try {
        const parsed = math.parse(expr);
        // Compile expression
        mathCompiledExpression = parsed.compile();
        
        // Test compile with a arbitrary x value to guarantee standard function behavior
        const testVal = mathCompiledExpression.evaluate({ x: 1 });
        if (typeof testVal === 'object' && testVal !== null && testVal.isComplex) {
          showError("Formula outputs complex numbers. Please use real expressions.");
          return false;
        }

        // Hide error state, show success
        functionStatus.classList.remove('hidden');
        functionError.classList.add('hidden');
        return true;
      } catch (err) {
        showError(err.message || "Syntactical error in mathematical formula.");
        return false;
      }
    }

    function showError(msg) {
      functionStatus.classList.add('hidden');
      functionError.classList.remove('hidden');
      errorMessage.textContent = msg;
      mathCompiledExpression = null;
    }

    // Safely evaluate Compiled Math.js Expression
    function evaluateF(x) {
      if (!mathCompiledExpression) return 0;
      try {
        const result = mathCompiledExpression.evaluate({ x: x });
        if (typeof result === 'number' && !isNaN(result) && isFinite(result)) {
          return result;
        }
        return 0; // fallback safe evaluation
      } catch (e) {
        return 0; // fallback
      }
    }

    // Safely parse numerical values / expressions (e.g., pi/2 or sqrt(3))
    function parseValueSafe(inputVal, fallback) {
      try {
        const parsed = math.evaluate(inputVal);
        if (typeof parsed === 'number' && !isNaN(parsed) && isFinite(parsed)) {
          return parsed;
        }
        return fallback;
      } catch(e) {
        return fallback;
      }
    }

    // Clean decimals for display
    function formatValue(val) {
      if (Math.abs(val) < 1e-4 && val !== 0) {
        return val.toExponential(4);
      }
      return val.toLocaleString(undefined, { minimumFractionDigits: 5, maximumFractionDigits: 5 });
    }

    // Approximate Integrals using various rules
    function runSimpsonRule(a, b, N) {
      if (N % 2 !== 0) N += 1; // Simpson's rule requires even number of intervals
      const h = (b - a) / N;
      let sum = evaluateF(a) + evaluateF(b);
      
      for (let i = 1; i < N; i++) {
        const x = a + i * h;
        if (i % 2 === 0) {
          sum += 2 * evaluateF(x);
        } else {
          sum += 4 * evaluateF(x);
        }
      }
      return (h / 3) * sum;
    }

    function runTrapezoidalRule(a, b, N) {
      const h = (b - a) / N;
      let sum = 0.5 * (evaluateF(a) + evaluateF(b));
      
      for (let i = 1; i < N; i++) {
        const x = a + i * h;
        sum += evaluateF(x);
      }
      return h * sum;
    }

    function runMidpointSum(a, b, N) {
      const h = (b - a) / N;
      let sum = 0;
      
      for (let i = 0; i < N; i++) {
        const midX = a + (i + 0.5) * h;
        sum += evaluateF(midX);
      }
      return h * sum;
    }

    function runLeftRiemannSum(a, b, N) {
      const h = (b - a) / N;
      let sum = 0;
      
      for (let i = 0; i < N; i++) {
        const x = a + i * h;
        sum += evaluateF(x);
      }
      return h * sum;
    }

    function runRightRiemannSum(a, b, N) {
      const h = (b - a) / N;
      let sum = 0;
      
      for (let i = 1; i <= N; i++) {
        const x = a + i * h;
        sum += evaluateF(x);
      }
      return h * sum;
    }

    // Precise Derivative Approximation at point x
    function approxDerivative(x, h = 1e-5) {
      return (evaluateF(x + h) - evaluateF(x - h)) / (2 * h);
    }

    // Compute Arc Length: Integral_a^b ( sqrt(1 + f'(x)^2) dx )
    function runArcLengthCalculation(a, b, N = 100) {
      if (N % 2 !== 0) N += 1;
      const h = (b - a) / N;
      
      const arcIntegrand = (x) => {
        const df = approxDerivative(x);
        return Math.sqrt(1 + df * df);
      };

      let sum = arcIntegrand(a) + arcIntegrand(b);
      for (let i = 1; i < N; i++) {
        const x = a + i * h;
        if (i % 2 === 0) {
          sum += 2 * arcIntegrand(x);
        } else {
          sum += 4 * arcIntegrand(x);
        }
      }
      return (h / 3) * sum;
    }

    // Perform entire set of calculations and update text
    function performCalculations() {
      if (!mathCompiledExpression) return;

      const a = parseValueSafe(boundAInput.value, -1);
      const b = parseValueSafe(boundBInput.value, 3);
      const N = parseInt(segmentsSlider.value) || 100;

      // Simpson is considered the primary / best value in this tool
      const simpsonResult = runSimpsonRule(a, b, N);
      const trapezoidResult = runTrapezoidalRule(a, b, N);
      const midpointResult = runMidpointSum(a, b, N);
      const leftResult = runLeftRiemannSum(a, b, N);
      const rightResult = runRightRiemannSum(a, b, N);

      // Display main results
      integralResultValue.textContent = formatValue(simpsonResult);
      
      // Arc Length
      const lengthResult = runArcLengthCalculation(a, b, N);
      arcLengthValueDisplay.textContent = formatValue(lengthResult);

      // Average Value: 1/(b-a) * Integral
      const width = b - a;
      if (Math.abs(width) > 1e-8) {
        const averageVal = simpsonResult / width;
        avgValueDisplay.textContent = formatValue(averageVal);
      } else {
        avgValueDisplay.textContent = "N/A";
      }

      // Format formula display markers
      mathDisplayA.textContent = a.toLocaleString(undefined, { maximumFractionDigits: 3 });
      mathDisplayB.textContent = b.toLocaleString(undefined, { maximumFractionDigits: 3 });
      mathDisplayFx.textContent = functionInput.value;

      // Manage Area signs
      if (simpsonResult > 0.0001) {
        areaSignDisplay.innerHTML = `<i data-lucide-trending-up class="w-3.5 h-3.5"></i> Positive Area`;
        areaSignDisplay.className = "text-xs font-semibold inline-flex items-center gap-1 mt-0.5 px-2.5 py-0.5 rounded-full bg-emerald-500/20 text-emerald-300";
      } else if (simpsonResult < -0.0001) {
        areaSignDisplay.innerHTML = `<i data-lucide-trending-down class="w-3.5 h-3.5"></i> Negative Area`;
        areaSignDisplay.className = "text-xs font-semibold inline-flex items-center gap-1 mt-0.5 px-2.5 py-0.5 rounded-full bg-rose-500/20 text-rose-300";
      } else {
        areaSignDisplay.innerHTML = `<i data-lucide-minus-circle class="w-3.5 h-3.5"></i> Net Zero Area`;
        areaSignDisplay.className = "text-xs font-semibold inline-flex items-center gap-1 mt-0.5 px-2.5 py-0.5 rounded-full bg-slate-500/20 text-slate-300";
      }
      lucide.createIcons();

      // Populate Comparison Table & percentage variance relative to Simpson's rule
      const ref = simpsonResult;
      const getVarianceStr = (val) => {
        if (Math.abs(ref) < 1e-6) return "—";
        const pctDiff = ((val - ref) / Math.abs(ref)) * 100;
        const sign = pctDiff > 0 ? "+" : "";
        return `${sign}${pctDiff.toFixed(3)}%`;
      };

      document.getElementById('calc-val-simpson').textContent = formatValue(simpsonResult);
      document.getElementById('err-simpson').textContent = "Ref (Best)";

      document.getElementById('calc-val-trapezoid').textContent = formatValue(trapezoidResult);
      document.getElementById('err-trapezoid').textContent = getVarianceStr(trapezoidResult);

      document.getElementById('calc-val-midpoint').textContent = formatValue(midpointResult);
      document.getElementById('err-midpoint').textContent = getVarianceStr(midpointResult);

      document.getElementById('calc-val-left').textContent = formatValue(leftResult);
      document.getElementById('err-left').textContent = getVarianceStr(leftResult);

      document.getElementById('calc-val-right').textContent = formatValue(rightResult);
      document.getElementById('err-right').textContent = getVarianceStr(rightResult);
    }

    // Charting functionality using Chart.js
    function renderChart() {
      if (!mathCompiledExpression) return;

      const a = parseValueSafe(boundAInput.value, -1);
      const b = parseValueSafe(boundBInput.value, 3);

      // Determine view range with dynamic cushion margins around a and b
      const width = Math.abs(b - a);
      const minBound = Math.min(a, b);
      const maxBound = Math.max(a, b);
      const cushion = width > 0 ? Math.max(width * 0.4, 1.5) : 3;
      
      const xStart = minBound - cushion;
      const xEnd = maxBound + cushion;

      const totalPlotPoints = 250;
      const step = (xEnd - xStart) / totalPlotPoints;

      const fullCurveData = [];
      const shadedAreaData = [];

      // Generate points for full curve and check singularity values
      for (let i = 0; i <= totalPlotPoints; i++) {
        const x = xStart + i * step;
        const y = evaluateF(x);
        
        // Skip extreme infinity values to keep graph clean
        if (Math.abs(y) < 1000) {
          fullCurveData.push({ x: x, y: y });
        }
      }

      // Generate points inside integrated region a -> b
      // Density in shaded area can be higher for smoother visual edges
      const shadeSteps = 150;
      const shadeStepSize = (b - a) / shadeSteps;
      for (let i = 0; i <= shadeSteps; i++) {
        const x = a + i * shadeStepSize;
        const y = evaluateF(x);
        if (Math.abs(y) < 1000) {
          shadedAreaData.push({ x: x, y: y });
        }
      }

      // Boundary Line datasets
      const verticalLineA = [
        { x: a, y: 0 },
        { x: a, y: evaluateF(a) }
      ];
      const verticalLineB = [
        { x: b, y: 0 },
        { x: b, y: evaluateF(b) }
      ];

      // Build or update the Chart
      const ctx = document.getElementById('integrationChart').getContext('2d');
      
      if (chartInstance) {
        chartInstance.destroy();
      }

      chartInstance = new Chart(ctx, {
        type: 'line',
        data: {
          datasets: [
            {
              label: 'f(x)',
              data: fullCurveData,
              borderColor: '#4f46e5', // indigo-600
              borderWidth: 3,
              pointRadius: 0,
              hoverRadius: 0,
              fill: false,
              tension: 0.15,
              z: 20
            },
            {
              label: 'Shaded Area',
              data: shadedAreaData,
              borderColor: 'transparent',
              backgroundColor: 'rgba(79, 70, 229, 0.22)', // semi-trans indigo
              fill: 'origin',
              pointRadius: 0,
              hoverRadius: 0,
              z: 10
            },
            {
              label: 'Bound a',
              data: verticalLineA,
              borderColor: '#ef4444', // red-500
              borderWidth: 2,
              borderDash: [5, 4],
              pointRadius: 4,
              pointBackgroundColor: '#ef4444',
              fill: false,
              z: 30
            },
            {
              label: 'Bound b',
              data: verticalLineB,
              borderColor: '#10b981', // emerald-500
              borderWidth: 2,
              borderDash: [5, 4],
              pointRadius: 4,
              pointBackgroundColor: '#10b981',
              fill: false,
              z: 30
            }
          ]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          plugins: {
            legend: {
              display: false // We use custom legends
            },
            tooltip: {
              enabled: true,
              mode: 'index',
              intersect: false,
              callbacks: {
                title: function(context) {
                  return `x = ${parseFloat(context[0].parsed.x).toFixed(4)}`;
                },
                label: function(context) {
                  if (context.dataset.label === 'f(x)') {
                    return `f(x) = ${parseFloat(context.parsed.y).toFixed(4)}`;
                  }
                  return null;
                }
              }
            }
          },
          scales: {
            x: {
              type: 'linear',
              position: 'bottom',
              grid: {
                display: gridVisible,
                color: 'rgba(226, 232, 240, 0.8)' // slate-200
              },
              title: {
                display: true,
                text: 'X Axis',
                font: {
                  size: 11,
                  weight: 'semibold'
                }
              }
            },
            y: {
              type: 'linear',
              grid: {
                display: gridVisible,
                color: 'rgba(226, 232, 240, 0.8)'
              },
              title: {
                display: true,
                text: 'Y Axis',
                font: {
                  size: 11,
                  weight: 'semibold'
                }
              }
            }
          },
          animation: {
            duration: autoScaleOnUpdate ? 600 : 0 // Smooth transition animations
          }
        }
      });
      
      // AutoScale on update setting resets to active after initial load animation
      autoScaleOnUpdate = true;
    }

    // Handles synchronization when bounds are updated
    function updateBoundsState(source) {
      let a = parseValueSafe(boundAInput.value, -1);
      let b = parseValueSafe(boundBInput.value, 3);

      if (source === 'input-a') {
        boundASlider.value = a;
        boundAValDisplay.textContent = a;
      } else if (source === 'slider-a') {
        a = parseFloat(boundASlider.value);
        boundAInput.value = a;
        boundAValDisplay.textContent = a;
      } else if (source === 'input-b') {
        boundBSlider.value = b;
        boundBValDisplay.textContent = b;
      } else if (source === 'slider-b') {
        b = parseFloat(boundBSlider.value);
        boundBInput.value = b;
        boundBValDisplay.textContent = b;
      }

      performCalculations();
      renderChart();
    }
